add_library(llama-glm-wire STATIC stage-crypto.cpp stage-wire.cpp stage-auth.cpp)
target_include_directories(llama-glm-wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(llama-glm-wire PUBLIC cxx_std_17)
if (WIN32)
    target_link_libraries(llama-glm-wire PRIVATE bcrypt)
else()
    find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
    target_link_libraries(llama-glm-wire PRIVATE OpenSSL::Crypto)
endif()
